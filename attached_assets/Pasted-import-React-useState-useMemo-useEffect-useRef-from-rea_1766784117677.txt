import React, { useState, useMemo, useEffect, useRef } from 'react';
import { 
  Shield, Cpu, TrendingUp, Briefcase, Coins, Globe, 
  Search, Sun, Moon, ChevronRight, Filter, Lock, 
  CheckCircle, X, Share2, Bookmark, ExternalLink, 
  Server, Loader2, FileText, Sparkles, Volume2, 
  Radio, RefreshCw, Zap
} from 'lucide-react';

// --- Configuration & Constants ---
const apiKey = ""; 
const UPDATE_INTERVAL_MS = 300000; // 5 Minutes
const DEMO_MODE_MS = 10000;       // 10 Seconds (for testing)

const CATEGORIES = [
  { id: 'all', label: 'Live Intel', icon: Globe },
  { id: 'Cybersecurity', label: 'Security', icon: Shield },
  { id: 'Tech', label: 'Tech', icon: Cpu },
  { id: 'Crypto', label: 'Crypto', icon: Coins },
];

// --- Mock Content Engine ---
const FEED_TEMPLATES = [
  { category: "Cybersecurity", source: "BleepingComputer", threatLevel: "High", titles: ["New Ransomware Variant 'DarkVault'", "Zero-Day in Enterprise VPN", "AI-Voice Phishing Spike"] },
  { category: "Crypto", source: "CoinDesk", threatLevel: "Medium", titles: ["Exchange Liquidity Warning", "DeFi Flash Loan Exploit", "Regulatory Shift in EU"] },
  { category: "Tech", source: "The Verge", threatLevel: "Low", titles: ["Quantum Chip Breakthrough", "Cloud Infrastructure Outage", "Next-Gen AI Hardware"] }
];

const generateArticle = () => {
  const template = FEED_TEMPLATES[Math.floor(Math.random() * FEED_TEMPLATES.length)];
  const title = template.titles[Math.floor(Math.random() * template.titles.length)];
  return {
    id: Date.now() + Math.random(),
    title: `${title} - ${new Date().toLocaleTimeString([], {hour: '2-digit', minute:'2-digit', second:'2-digit'})}`,
    source: template.source,
    time: "Just Now",
    category: template.category,
    threatLevel: template.threatLevel,
    imageUrl: `https://picsum.photos/seed/${Math.random()}/800/500`,
    excerpt: `Real-time intelligence report regarding ${title.toLowerCase()}. Automated system detected high-velocity activity in this sector.`,
    url: "https://gem-assist.security",
    content: ["Automated monitoring has flagged this event for immediate review.", "Recommended action: Verify perimeter logs."]
  };
};

// --- Components ---

const NewsCard = ({ article, onClick }) => (
  <div 
    onClick={() => onClick(article)}
    className="group bg-white dark:bg-slate-900 rounded-xl overflow-hidden border border-slate-200 dark:border-slate-800 hover:border-emerald-500/50 transition-all cursor-pointer animate-in slide-in-from-top-4 duration-500"
  >
    <div className="relative h-40 overflow-hidden">
      <img src={article.imageUrl} alt="" className="w-full h-full object-cover group-hover:scale-105 transition-transform duration-700" />
      <div className="absolute top-2 left-2 flex gap-1">
        <span className="bg-slate-900/80 backdrop-blur px-2 py-0.5 rounded text-[10px] font-bold text-white uppercase">{article.category}</span>
        {article.threatLevel === 'High' && <span className="bg-red-600 px-2 py-0.5 rounded text-[10px] font-bold text-white animate-pulse">HIGH RISK</span>}
      </div>
    </div>
    <div className="p-4">
      <div className="flex items-center gap-2 mb-2 text-[10px] font-bold text-slate-500 uppercase tracking-widest">
        <span>{article.source}</span>
        <span>•</span>
        <span className="text-emerald-500">{article.time}</span>
      </div>
      <h3 className="font-bold text-slate-900 dark:text-white leading-tight mb-2 group-hover:text-emerald-400 transition-colors line-clamp-2">{article.title}</h3>
      <p className="text-xs text-slate-500 dark:text-slate-400 line-clamp-2">{article.excerpt}</p>
    </div>
  </div>
);

export default function App() {
  const [isDarkMode, setIsDarkMode] = useState(true);
  const [feed, setFeed] = useState([generateArticle(), generateArticle()]);
  const [activeTab, setActiveTab] = useState('all');
  const [isDemoMode, setIsDemoMode] = useState(true); // Default to fast for visibility
  const [lastUpdate, setLastUpdate] = useState(new Date());

  // THE AUTO-UPLOAD ENGINE
  useEffect(() => {
    const intervalTime = isDemoMode ? DEMO_MODE_MS : UPDATE_INTERVAL_MS;
    
    const timer = setInterval(() => {
      const newStory = generateArticle();
      // Auto-upload to top of feed
      setFeed(prev => [newStory, ...prev].slice(0, 20)); // Keep last 20
      setLastUpdate(new Date());
    }, intervalTime);

    return () => clearInterval(timer);
  }, [isDemoMode]);

  const filteredFeed = useMemo(() => {
    return activeTab === 'all' ? feed : feed.filter(a => a.category === activeTab);
  }, [feed, activeTab]);

  return (
    <div className={`h-[800px] w-full flex flex-col font-sans transition-colors ${isDarkMode ? 'dark bg-slate-950' : 'bg-slate-50'}`}>
      
      {/* Header */}
      <header className="h-16 border-b border-slate-200 dark:border-slate-800 bg-white/80 dark:bg-slate-900/80 backdrop-blur-md flex items-center justify-between px-6 sticky top-0 z-50">
        <div className="flex items-center gap-3">
          <div className="bg-emerald-600 p-1.5 rounded-lg shadow-lg shadow-emerald-500/20">
            <Shield size={20} className="text-white" />
          </div>
          <div>
            <h1 className="text-sm font-black dark:text-white uppercase tracking-tighter">GEM INTEL <span className="text-emerald-500">AUTO-FEED</span></h1>
            <div className="flex items-center gap-1.5">
              <div className="w-1.5 h-1.5 bg-emerald-500 rounded-full animate-ping"></div>
              <span className="text-[9px] font-mono text-emerald-400 uppercase tracking-widest">Live Uplink Active</span>
            </div>
          </div>
        </div>

        <div className="flex items-center gap-4">
          <button 
            onClick={() => setIsDemoMode(!isDemoMode)}
            className={`text-[10px] font-bold px-3 py-1 rounded-full border transition-all ${isDemoMode ? 'bg-amber-500/10 border-amber-500/50 text-amber-500' : 'bg-slate-800 border-slate-700 text-slate-400'}`}
          >
            {isDemoMode ? '🚀 TEST MODE (10s)' : '⏱️ PROD MODE (5m)'}
          </button>
          <button onClick={() => setIsDarkMode(!isDarkMode)} className="text-slate-400 hover:text-white">
            {isDarkMode ? <Sun size={18} /> : <Moon size={18} />}
          </button>
        </div>
      </header>

      {/* Nav */}
      <nav className="flex gap-2 p-4 overflow-x-auto bg-white dark:bg-slate-950 border-b border-slate-200 dark:border-slate-800">
        {CATEGORIES.map(cat => (
          <button
            key={cat.id}
            onClick={() => setActiveTab(cat.id)}
            className={`flex items-center gap-2 px-4 py-1.5 rounded-full text-xs font-bold transition-all whitespace-nowrap ${
              activeTab === cat.id ? 'bg-emerald-600 text-white shadow-lg' : 'bg-slate-100 dark:bg-slate-800 text-slate-500 hover:bg-slate-200'
            }`}
          >
            <cat.icon size={14} /> {cat.label}
          </button>
        ))}
      </nav>

      {/* Content */}
      <main className="flex-1 overflow-y-auto p-6">
        <div className="max-w-5xl mx-auto">
          
          <div className="flex justify-between items-end mb-8">
            <div>
              <h2 className="text-2xl font-black dark:text-white">Threat Stream</h2>
              <p className="text-xs text-slate-500">Last synchronized: {lastUpdate.toLocaleTimeString()}</p>
            </div>
            <div className="flex items-center gap-2 text-[10px] font-mono text-slate-500">
              <RefreshCw size={10} className="animate-spin-slow" /> AUTO-SYNC ENABLED
            </div>
          </div>

          <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-6">
            {filteredFeed.map(article => (
              <NewsCard key={article.id} article={article} onClick={(a) => console.log("Open", a)} />
            ))}
          </div>
        </div>
      </main>

      {/* Footer Status */}
      <footer className="h-10 bg-slate-100 dark:bg-slate-900 border-t border-slate-200 dark:border-slate-800 px-6 flex items-center justify-between text-[10px] font-mono text-slate-500">
        <div className="flex items-center gap-4">
          <span className="flex items-center gap-1"><Zap size={10} className="text-emerald-500" /> SYSTEM: ONLINE</span>
          <span className="flex items-center gap-1"><Lock size={10} /> ENCRYPTION: AES-256</span>
        </div>
        <div>TOTAL INTEL: {feed.length} EVENTS</div>
      </footer>
    </div>
  );
}
