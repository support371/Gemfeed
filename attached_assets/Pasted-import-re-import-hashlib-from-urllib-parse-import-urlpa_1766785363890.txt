import re
import hashlib
from urllib.parse import urlparse, urlencode, urlunparse, parse_qsl

BANNED_NEXTDOOR_TERMS = {
    "hodl", "bullish", "moon", "to the moon", "pump", "dump", "ape in", "lambo"
}

MAX_NEXTDOOR_CHARS = 900  # keep it short; tune as needed
MAX_HASHTAGS = 0          # Nextdoor: safest is none

def add_utm(url: str, source: str, medium: str = "social", campaign: str = "gem_intel") -> str:
    """Adds/overwrites UTM params safely."""
    if not url:
        return url
    parts = urlparse(url)
    q = dict(parse_qsl(parts.query))
    q.update({
        "utm_source": source,
        "utm_medium": medium,
        "utm_campaign": campaign,
    })
    new_query = urlencode(q)
    return urlunparse((parts.scheme, parts.netloc, parts.path, parts.params, new_query, parts.fragment))

def sanitize_nextdoor_text(text: str) -> str:
    """Remove slang, reduce hype, minimize spam signals."""
    if not text:
        return ""

    t = text.strip()

    # Remove hashtags
    t = re.sub(r"#\w+", "", t)

    # Remove excessive punctuation / hype
    t = re.sub(r"[!]{2,}", "!", t)
    t = re.sub(r"[\U0001F300-\U0001FAFF]+", "", t)  # optional: strip emoji blocks if too spammy

    # Normalize whitespace
    t = re.sub(r"\s+", " ", t).strip()

    # Ban crypto-slang terms
    lowered = t.lower()
    for term in BANNED_NEXTDOOR_TERMS:
        if term in lowered:
            # Remove the term (soft approach) rather than failing hard
            pattern = re.compile(re.escape(term), re.IGNORECASE)
            t = pattern.sub("", t)
            lowered = t.lower()

    # Hard limit length
    if len(t) > MAX_NEXTDOOR_CHARS:
        t = t[:MAX_NEXTDOOR_CHARS].rsplit(" ", 1)[0].strip() + "…"

    return t

def build_nextdoor_caption(title: str, tip: str, benefit: str, url: str) -> str:
    """
    Nextdoor tone: local utility, neighbor-friendly, no hype.
    Keep 2–5 sentences.
    """
    url = add_utm(url, source="nextdoor")
    lines = [
        "Hello neighbors — quick security & savings tip:",
        f"{title}".strip(),
        f"{tip}".strip(),
        f"{benefit}".strip(),
        f"More info: {url}".strip()
    ]
    caption = "\n".join([ln for ln in lines if ln])
    return sanitize_nextdoor_text(caption)

def build_ayrshare_payload_for_nextdoor(
    *,
    post_text: str,
    media_urls: list[str] | None = None,
    scope: str = "public",
) -> dict:
    """
    Builds a minimal, Nextdoor-friendly Ayrshare payload.
    NOTE: Keep platform list restricted here; upstream orchestrator can add others.
    """
    payload = {
        "post": sanitize_nextdoor_text(post_text),
        "platforms": ["nextdoor"],
        "nextdoorOptions": {"scope": scope},
    }
    if media_urls:
        payload["mediaUrls"] = media_urls[:1]  # safest: 1 media item for Nextdoor
    return payload

def stable_post_key(platform: str, article_id: str, url: str) -> str:
    """Use for dedupe store: platform + hash(article_id or url)."""
    base = f"{platform}:{article_id or url}"
    return hashlib.sha256(base.encode("utf-8")).hexdigest()
